import logging
import os
import re
from urllib.parse import urlparse

import httpx
from dotenv import load_dotenv
from telegram import ChatPermissions, Update
from telegram.constants import ChatType
from telegram.error import TelegramError
from telegram.ext import Application, CommandHandler, ContextTypes, MessageHandler, filters

load_dotenv("/home/psycedelic/Projects/psycedelicai-telegram-bot/.env")

TOKEN = os.getenv("TELEGRAM_BOT_TOKEN", "").strip()
BASE_URL = os.getenv("GPT4ALL_BASE_URL", "http://127.0.0.1:4891/v1").rstrip("/")
MODEL = os.getenv("GPT4ALL_MODEL", "Llama 3.2 3B Instruct")
OWNER_ID = int(os.getenv("ALLOWED_TELEGRAM_USER_ID", "0"))
LOG_CHAT_ID = os.getenv("MODERATOR_LOG_CHAT_ID", str(OWNER_ID)).strip()

MODERATOR_IDS = {
    int(x.strip())
    for x in os.getenv("MODERATOR_USER_IDS", "").split(",")
    if x.strip().isdigit()
}

AUTHORIZED_IDS = MODERATOR_IDS | {OWNER_ID}

BLOCKED_DOMAINS = {
    "test-blocked.invalid",
}

HIGH_CONFIDENCE_TERMS = set()

logging.basicConfig(
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    level=logging.INFO,
)

logger = logging.getLogger(__name__)
logging.getLogger("httpx").setLevel(logging.WARNING)

if not TOKEN:
    raise RuntimeError("TELEGRAM_BOT_TOKEN is missing")

if not OWNER_ID:
    raise RuntimeError("ALLOWED_TELEGRAM_USER_ID is missing")


def is_owner(update: Update) -> bool:
    user = update.effective_user
    return user is not None and user.id == OWNER_ID


def is_authorized(update: Update) -> bool:
    user = update.effective_user
    return user is not None and user.id in AUTHORIZED_IDS


def is_group(update: Update) -> bool:
    chat = update.effective_chat
    return chat is not None and chat.type in (
        ChatType.GROUP,
        ChatType.SUPERGROUP,
    )


async def deny(update: Update) -> None:
    if update.effective_message:
        await update.effective_message.reply_text("Access denied.")


def normalize(text: str) -> str:
    replacements = {
        "hxxp://": "http://",
        "hxxps://": "https://",
        "[.]": ".",
        "(.)": ".",
        "{.}": ".",
        " dot ": ".",
        "[:]": ":",
        "[://]": "://",
    }

    text = text.lower()

    for old, new in replacements.items():
        text = text.replace(old, new)

    text = re.sub(r"[\u200b-\u200f\u2060\ufeff]", "", text)
    return re.sub(r"\s+", " ", text).strip()


def extract_urls(text: str) -> list[str]:
    return re.findall(
        r"(?:https?://|www\.)[^\s<>'\"()\[\]{}]+",
        text,
        re.IGNORECASE,
    )


def domain_from_url(url: str) -> str:
    if url.lower().startswith("www."):
        url = "https://" + url

    if "://" not in url:
        url = "https://" + url

    try:
        return (urlparse(url).hostname or "").lower().rstrip(".")
    except ValueError:
        return ""


def is_blocked(domain: str) -> bool:
    return any(
        domain == blocked or domain.endswith("." + blocked)
        for blocked in BLOCKED_DOMAINS
    )


def classify(message) -> tuple[bool, list[str]]:
    text = normalize(
        (message.text or "") + "\n" + (message.caption or "")
    )

    urls = extract_urls(text)
    domains = [domain_from_url(url) for url in urls]
    reasons = []

    if any(is_blocked(domain) for domain in domains):
        reasons.append("blocked domain")

    if urls and any(term.lower() in text for term in HIGH_CONFIDENCE_TERMS):
        reasons.append("high-confidence indicator with URL")

    return bool(reasons), reasons


async def send_log(context, message, reasons, actions) -> None:
    if not LOG_CHAT_ID:
        return

    user = message.from_user
    name = user.full_name if user else "Unknown"
    text = (
        "Moderation report\n"
        f"Chat: {message.chat.title}\n"
        f"User: {name}\n"
        f"User ID: {user.id if user else 'unknown'}\n"
        f"Message ID: {message.message_id}\n"
        f"Reason: {', '.join(reasons)}\n"
        f"Action: {', '.join(actions)}"
    )

    try:
        await context.bot.send_message(
            chat_id=LOG_CHAT_ID,
            text=text,
            disable_web_page_preview=True,
        )
    except TelegramError:
        logger.exception("Could not send moderator report")


async def automatic_moderation(update: Update, context) -> None:
    message = update.effective_message

    if not message or not is_group(update):
        return

    if not message.from_user or message.from_user.is_bot:
        return

    should_act, reasons = classify(message)

    if not should_act:
        return

    actions = []

    try:
        await context.bot.delete_message(
            chat_id=message.chat.id,
            message_id=message.message_id,
        )
        actions.append("message deleted")
    except TelegramError:
        actions.append("deletion failed")

    try:
        await context.bot.ban_chat_member(
            chat_id=message.chat.id,
            user_id=message.from_user.id,
            revoke_messages=True,
        )
        actions.append("user banned")
    except TelegramError:
        actions.append("ban failed")

    await send_log(context, message, reasons, actions)


async def target(update: Update):
    if not is_authorized(update):
        await deny(update)
        return None

    if not is_group(update):
        await update.effective_message.reply_text(
            "This command only works in a group."
        )
        return None

    message = update.effective_message
    target_message = message.reply_to_message

    if not target_message or not target_message.from_user:
        await message.reply_text(
            "Reply to the message you want to moderate."
        )
        return None

    if target_message.from_user.id == OWNER_ID:
        await message.reply_text(
            "The bot owner cannot be moderated."
        )
        return None

    return target_message


async def start(update: Update, context) -> None:
    if not is_owner(update):
        await deny(update)
        return

    await update.effective_message.reply_text(
        "PsycedelicAI bot is online."
    )


async def help_command(update: Update, context) -> None:
    if not is_authorized(update):
        await deny(update)
        return

    await update.effective_message.reply_text(
        "Private AI chat is available to the owner.\n\n"
        "Moderation commands:\n"
        "/delete /warn /mute /unmute\n"
        "/kick /ban /report /status /rules\n"
        "/unban USER_ID"
    )


async def delete_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await message.delete()
        await update.effective_message.reply_text("Message deleted.")


async def warn_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await update.effective_message.reply_text(
            f"Warning issued to {message.from_user.full_name}."
        )


async def mute_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await update.effective_chat.restrict_member(
            user_id=message.from_user.id,
            permissions=ChatPermissions(can_send_messages=False),
        )
        await update.effective_message.reply_text("User muted.")


async def unmute_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await update.effective_chat.restrict_member(
            user_id=message.from_user.id,
            permissions=ChatPermissions.all_permissions(),
        )
        await update.effective_message.reply_text("User unmuted.")


async def kick_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await update.effective_chat.ban_member(message.from_user.id)
        await update.effective_chat.unban_member(message.from_user.id)
        await update.effective_message.reply_text("User removed.")


async def ban_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await update.effective_chat.ban_member(message.from_user.id)
        await update.effective_message.reply_text("User banned.")


async def report_command(update: Update, context) -> None:
    message = await target(update)
    if message:
        await context.bot.send_message(
            chat_id=LOG_CHAT_ID,
            text=(
                "Manual moderation report\n"
                f"Group: {update.effective_chat.title}\n"
                f"User ID: {message.from_user.id}\n"
                f"Message ID: {message.message_id}"
            ),
        )
        await update.effective_message.reply_text("Report sent.")


async def status_command(update: Update, context) -> None:
    if not is_authorized(update):
        await deny(update)
        return

    await update.effective_message.reply_text(
        f"Bot online.\nBlocked domains: {len(BLOCKED_DOMAINS)}\n"
        f"Moderators: {len(MODERATOR_IDS)}"
    )


async def rules_command(update: Update, context) -> None:
    if not is_authorized(update):
        await deny(update)
        return

    await update.effective_message.reply_text(
        "Blocked domains are deleted automatically. "
        "Manual commands require replying to a message."
    )


async def unban_command(update: Update, context) -> None:
    if not is_authorized(update):
        await deny(update)
        return

    if not context.args:
        await update.effective_message.reply_text(
            "Usage: /unban USER_ID"
        )
        return

    try:
        await update.effective_chat.unban_member(int(context.args[0]))
        await update.effective_message.reply_text("User unbanned.")
    except (ValueError, TelegramError):
        await update.effective_message.reply_text("Unban failed.")


async def ask_ai(text: str) -> str:
    async with httpx.AsyncClient(timeout=120) as client:
        response = await client.post(
            f"{BASE_URL}/chat/completions",
            json={
                "model": MODEL,
                "messages": [
                    {
                        "role": "system",
                        "content": (
                            "You are PsycedelicAI. "
                            "Always answer in English."
                        ),
                    },
                    {"role": "user", "content": text},
                ],
                "temperature": 0.7,
                "max_tokens": 500,
            },
        )
        response.raise_for_status()
        return response.json()["choices"][0]["message"]["content"].strip()


async def text_message(update: Update, context) -> None:
    if is_group(update):
        await automatic_moderation(update, context)
        return

    if not is_owner(update):
        await deny(update)
        return

    message = update.effective_message

    if not message or not message.text:
        return

    await update.effective_chat.send_action("typing")

    try:
        await message.reply_text(await ask_ai(message.text))
    except httpx.ConnectError:
        await message.reply_text("Could not connect to GPT4All.")
    except Exception:
        logger.exception("AI request failed")
        await message.reply_text("AI request failed.")


def main() -> None:
    application = Application.builder().token(TOKEN).build()

    commands = {
        "start": start,
        "help": help_command,
        "delete": delete_command,
        "warn": warn_command,
        "mute": mute_command,
        "unmute": unmute_command,
        "kick": kick_command,
        "ban": ban_command,
        "report": report_command,
        "status": status_command,
        "rules": rules_command,
        "unban": unban_command,
    }

    for name, handler in commands.items():
        application.add_handler(CommandHandler(name, handler))

    application.add_handler(
        MessageHandler(
            filters.TEXT & ~filters.COMMAND,
            text_message,
        )
    )

    logger.info("Telegram bot starting")
    logger.info("GPT4All URL: %s", BASE_URL)
    logger.info("GPT4All model: %s", MODEL)

    application.run_polling()


if __name__ == "__main__":
    main()
